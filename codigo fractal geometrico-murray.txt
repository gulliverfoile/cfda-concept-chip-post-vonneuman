# src/generators/murray_fractal.py
"""
Generador de redes de bifurcación jerárquica basado en la Ley de Hess-Murray.
Optimizado para intercambiadores de calor y estructuras ligeras.
"""

import numpy as np
import trimesh
from scipy.spatial import distance

class MurrayFractalGenerator:
    """
    Genera una red de tubos que se bifurca siguiendo la Ley de Murray:
        r_padre^3 = r_hijo1^3 + r_hijo2^3

    Parámetros:
        outer_radius (float): radio de la esfera envolvente (mm)
        iterations (int): número de niveles de bifurcación (0 = solo el tronco)
        min_beam_radius (float): radio mínimo de las vigas (mm)
        min_angle (float): ángulo mínimo entre bifurcaciones (grados)
        seed (int): semilla para reproducibilidad
    """
    def __init__(self, outer_radius=1.0, iterations=3, min_beam_radius=0.05,
                 min_angle=15, seed=None):
        self.R = outer_radius
        self.iterations = iterations
        self.min_radius = min_beam_radius
        self.min_angle = np.deg2rad(min_angle)
        self.seed = seed
        if seed is not None:
            np.random.seed(seed)

        # Almacenamiento de nodos y aristas
        self.nodes = []          # Lista de (x, y, z)
        self.edges = []          # Lista de (idx1, idx2, r_start, r_end)
        self.radii = {}          # Diccionario: idx_nodo -> radio (para el tronco)

    def generate(self):
        """Genera la red completa y devuelve nodos y aristas."""
        # Nodo raíz en el centro
        self.nodes = [(0.0, 0.0, 0.0)]
        self.radii[0] = self.R * 0.3  # Radio inicial (se ajustará)

        # Generar recursivamente
        self._branch(0, self.radii[0], self.R, 0)

        # Convertir aristas a formato de vigas con tapering
        beams = []
        for i, j, r_start, r_end in self.edges:
            beams.append({
                'start': np.array(self.nodes[i]),
                'end': np.array(self.nodes[j]),
                'radius_start': r_start,
                'radius_end': r_end,
                'mean_radius': (r_start + r_end) / 2
            })

        return {
            'points': np.array(self.nodes),
            'beams': beams
        }

    def _branch(self, parent_idx, parent_radius, max_radius, level):
        """Crea dos hijos desde un nodo padre."""
        if level >= self.iterations:
            return

        # Calcular radio de los hijos según Murray: r_padre^3 = r1^3 + r2^3
        # Asumimos que los dos hijos tienen el mismo radio (distribución simétrica)
        r_child = parent_radius / (2 ** (1/3))
        if r_child < self.min_radius:
            return

        # Generar direcciones para los dos hijos
        dirs = self._generate_branch_directions(level)

        for direction in dirs:
            # Calcular nueva posición (a una distancia proporcional al radio)
            length = max_radius * 0.5 * (1 - level / (self.iterations + 1))
            new_pos = np.array(self.nodes[parent_idx]) + direction * length

            # Mantener dentro de la esfera
            norm = np.linalg.norm(new_pos)
            if norm > self.R:
                new_pos = new_pos * self.R / norm

            # Añadir nodo
            child_idx = len(self.nodes)
            self.nodes.append(tuple(new_pos))
            self.radii[child_idx] = r_child

            # Añadir arista (con tapering)
            self.edges.append((parent_idx, child_idx, parent_radius, r_child))

            # Recursión
            self._branch(child_idx, r_child, max_radius - length, level + 1)

    def _generate_branch_directions(self, level):
        """Genera dos direcciones con ángulo mínimo de separación."""
        # Dirección inicial aleatoria en la esfera
        theta = np.random.uniform(0, 2*np.pi)
        phi = np.arccos(2*np.random.uniform(0, 1) - 1)
        dir1 = np.array([
            np.sin(phi) * np.cos(theta),
            np.sin(phi) * np.sin(theta),
            np.cos(phi)
        ])

        # Rotar dir1 para obtener dir2 con el ángulo mínimo
        # Tomamos un vector perpendicular a dir1
        if np.abs(dir1[0]) < 0.9:
            ref = np.array([1, 0, 0])
        else:
            ref = np.array([0, 1, 0])
        v = np.cross(dir1, ref)
        v = v / np.linalg.norm(v)

        # Rotar dir1 alrededor de v un ángulo aleatorio entre min_angle y 45°
        angle = np.random.uniform(self.min_angle, np.pi/4)
        dir2 = self._rotate_vector(dir1, v, angle)

        # Asegurar que el ángulo entre dir1 y dir2 sea >= min_angle
        if np.arccos(np.clip(np.dot(dir1, dir2), -1, 1)) < self.min_angle:
            dir2 = -dir2

        return [dir1, dir2]

    def _rotate_vector(self, vec, axis, angle):
        """Rota un vector alrededor de un eje usando la fórmula de Rodrigues."""
        k = axis / np.linalg.norm(axis)
        v = vec
        return v * np.cos(angle) + np.cross(k, v) * np.sin(angle) + k * np.dot(k, v) * (1 - np.cos(angle))

    def calculate_properties(self, points, beams):
        """Calcula propiedades (porosidad, área superficial, etc.)."""
        total_volume = 4/3 * np.pi * self.R**3
        beam_volume = 0
        surface_area = 0

        for beam in beams:
            start, end = beam['start'], beam['end']
            length = np.linalg.norm(end - start)
            r1, r2 = beam['radius_start'], beam['radius_end']

            # Volumen de cono truncado
            beam_volume += np.pi/3 * length * (r1**2 + r1*r2 + r2**2)

            # Área lateral del cono truncado
            lateral_area = np.pi * (r1 + r2) * np.sqrt((r1 - r2)**2 + length**2)
            surface_area += lateral_area

        porosity = 1 - (beam_volume / total_volume) if total_volume > 0 else 0

        return {
            'porosity': porosity,
            'beam_count': len(beams),
            'surface_area': surface_area,
            'material_volume': beam_volume,
            'specific_surface': surface_area / beam_volume if beam_volume > 0 else 0
        }

    def export_to_stl(self, points, beams, filename):
        """Exporta a STL con cilindros y esferas en los nodos."""
        meshes = []

        for beam in beams:
            # Cilindro con tapering
            cyl = trimesh.creation.cylinder(
                radius=beam['mean_radius'],
                segment=[beam['start'], beam['end']],
                sections=8
            )
            meshes.append(cyl)

            # Esferas en extremos (para suavizar uniones)
            sphere1 = trimesh.creation.icosphere(
                radius=beam['radius_start'] * 1.1,
                center=beam['start']
            )
            sphere2 = trimesh.creation.icosphere(
                radius=beam['radius_end'] * 1.1,
                center=beam['end']
            )
            meshes.extend([sphere1, sphere2])

        # Combinar
        combined = trimesh.util.concatenate(meshes)
        combined.export(filename)
        print(f"STL exportado: {filename}")
        print(f"Triángulos: {len(combined.faces):,}")

    def generate_and_export(self, stl_filename=None):
        """Genera, calcula propiedades y exporta a STL."""
        data = self.generate()
        points = data['points']
        beams = data['beams']

        props = self.calculate_properties(points, beams)
        print("Propiedades del fractal de Murray:")
        for key, val in props.items():
            print(f"  {key}: {val:.4f}")

        if stl_filename:
            self.export_to_stl(points, beams, stl_filename)

        return data
